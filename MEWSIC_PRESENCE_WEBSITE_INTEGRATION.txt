BACKEND ENDPOINT: https://mewsic-presence.imrishan320.workers.dev

--------------------------------------------------------------------------------
1. OVERVIEW & OBJECTIVE
--------------------------------------------------------------------------------
Mewsic uses a Cloudflare Worker + D1 backend to monitor live active desktop app 
instances via a 5-minute heartbeat system.

The website agent should add a live active users indicator/badge (e.g., near the
Mewsic download button, hero section, or project card) on xeoniii.github.io to
showcase how many people are currently listening/using Mewsic.

--------------------------------------------------------------------------------
2. API SPECIFICATION
--------------------------------------------------------------------------------
- URL: https://mewsic-presence.imrishan320.workers.dev/active-users
- Method: GET
- Headers: None required
- CORS: Enabled (Access-Control-Allow-Origin: *)
- Response Schema (JSON):
  {
    "activeUsers": 1
  }

Notes:
- If the count is 0, activeUsers will be 0.
- If the endpoint cannot be reached or fails, handle gracefully (show fallback or hide badge).

--------------------------------------------------------------------------------
3. RECOMMENDED POLLING FREQUENCY
--------------------------------------------------------------------------------
- Recommended interval: Fetch on initial page load, then poll every 30 to 60 seconds.
- Cloudflare free tier allows 100,000 requests/day, so polling every 30-60s per visitor
  is well within free limits.

Add this JavaScript at the bottom of the page or in your main script:

<script>
(function initMewsicPresence() {
  const ENDPOINT = "https://mewsic-presence.imrishan320.workers.dev/active-users";
  const badgeEl = document.getElementById("mewsic-live-badge");
  const countEl = document.getElementById("mewsic-active-count");
  const labelEl = document.getElementById("mewsic-active-label");

  async function updateActiveUsers() {
    try {
      const response = await fetch(ENDPOINT, { cache: "no-store" });
      if (!response.ok) throw new Error("Network response not ok");
      const data = await response.json();
      
      const count = typeof data.activeUsers === "number" ? data.activeUsers : 0;
      
      if (badgeEl && countEl) {
        countEl.textContent = count;
        if (labelEl) {
          labelEl.textContent = count === 1 ? "user listening now" : "users listening now";
        }
        badgeEl.style.display = "inline-flex";
      }
    } catch (err) {
      // Graceful fallback: keep hidden or show neutral state if offline
      console.warn("Mewsic presence unreachable:", err);
    }
  }

  // Initial fetch
  updateActiveUsers();

  // Refresh every 30 seconds
  setInterval(updateActiveUsers, 30000);
})();
</script>


B. REACT / NEXT.JS COMPONENT (IF USING REACT)
----------------------------------------------
import React, { useEffect, useState } from "react";

export function MewsicLiveBadge() {
  const [activeUsers, setActiveUsers] = useState<number | null>(null);

  useEffect(() => {
    let isMounted = true;

    async function fetchUsers() {
      try {
        const res = await fetch("https://mewsic-presence.imrishan320.workers.dev/active-users", {
          cache: "no-store",
        });
        if (res.ok) {
          const data = await res.json();
          if (isMounted && typeof data.activeUsers === "number") {
            setActiveUsers(data.activeUsers);
          }
        }
      } catch (e) {
        // Silent catch
      }
    }

    fetchUsers();
    const interval = setInterval(fetchUsers, 30000);
    return () => {
      isMounted = false;
      clearInterval(interval);
    };
  }, []);

  if (activeUsers === null) return null;

  return (
    <div className="inline-flex items-center gap-2 px-3 py-1 text-xs font-medium text-emerald-400 bg-emerald-500/10 border border-emerald-500/20 rounded-full backdrop-blur-sm">
      <span className="relative flex h-2 w-2">
        <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-emerald-400 opacity-75"></span>
        <span className="relative inline-flex rounded-full h-2 w-2 bg-emerald-500"></span>
      </span>
      <span>
        {activeUsers} {activeUsers === 1 ? "user" : "users"} listening now
      </span>
    </div>
  );
}
